Integrasi Google Drive dengan Penamaan Cerdas

Kaidah Penamaan Cerdas (Baku RDOWS)

Sistem ini menggunakan algoritma pemotongan cerdas untuk membuang instruksi aktif dan mengubahnya menjadi frasa kata benda formal (Contoh: "Mengamati sosialisasi visi..." diubah menjadi "Bukti Observasi Fisik: Sosialisasi Visi..."). Selain itu, seluruh istilah "SPO" otomatis dikonversi menjadi "SOP".

1. Buat Proyek & Salin Kode

  • Buka web script.google.com (login akun Drive Klinik), klik Proyek Baru.
  • Klik Copy Code di bawah ini, lalu tempel ke layar putih yang tersedia.
function doPost(e) {
  var headers = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Methods": "POST, GET, OPTIONS", "Access-Control-Allow-Headers": "Content-Type" };
  try {
    if (!e || !e.postData || !e.postData.contents) return ContentService.createTextOutput(JSON.stringify({ success: false, error: "Data file tidak diterima." })).setMimeType(ContentService.MimeType.JSON).setHeaders(headers);
    var data = JSON.parse(e.postData.contents);
    if (data.action === 'delete') {
      var fileId = data.fileUrl.match(/[-\w]{25,}/);
      if (fileId && fileId[0]) { DriveApp.getFileById(fileId[0]).setTrashed(true); return ContentService.createTextOutput(JSON.stringify({success: true})).setMimeType(ContentService.MimeType.JSON).setHeaders(headers); } 
      else return ContentService.createTextOutput(JSON.stringify({success: false, error: "Invalid ID"})).setMimeType(ContentService.MimeType.JSON).setHeaders(headers);
    }
    if (!data.fileBase64) return ContentService.createTextOutput(JSON.stringify({ success: false, error: "File kosong." })).setMimeType(ContentService.MimeType.JSON).setHeaders(headers);

    var blob = Utilities.newBlob(Utilities.base64Decode(data.fileBase64.split(',')[1]), data.mimeType || "application/pdf", data.fileName || "Dokumen.pdf");
    var currentFolder = DriveApp.getRootFolder();
    var folders = (data.folderPath || "Akreditasi_Klinik").split('/');
    for (var i = 0; i < folders.length; i++) {
      if(folders[i] === "") continue;
      var iterator = currentFolder.getFoldersByName(folders[i]);
      if (iterator.hasNext()) currentFolder = iterator.next(); else currentFolder = currentFolder.createFolder(folders[i]);
    }
    var file = currentFolder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    return ContentService.createTextOutput(JSON.stringify({ success: true, url: file.getUrl(), fileName: file.getName() })).setMimeType(ContentService.MimeType.JSON).setHeaders(headers);
  } catch(error) { return ContentService.createTextOutput(JSON.stringify({ success: false, error: "CRASH SCRIPT: " + error.toString() })).setMimeType(ContentService.MimeType.JSON).setHeaders(headers); }
}
function doOptions(e) { return ContentService.createTextOutput("").setHeaders({"Access-Control-Allow-Origin": "*", "Access-Control-Allow-Methods": "POST, GET, OPTIONS", "Access-Control-Allow-Headers": "Content-Type"}); }

2. Terapkan (Deploy) Script

  • Klik Terapkan (Deploy) > Deployment baru > Roda Gigi ⚙️ > Aplikasi Web (Web App).
  • Jalankan sebagai: Saya (Me) | Siapa saja memiliki akses: Siapa saja (Anyone).
  • Klik Otorisasi akses > Lanjutan (Advanced) > Buka proyek (tidak aman) > Izinkan.
